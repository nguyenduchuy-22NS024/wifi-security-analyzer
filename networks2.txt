    def clean_text(text):
        if not text: return ""
        # Xóa các dấu hoa thị dư thừa ở đầu dòng (thường có trong iw scan)
        text = re.sub(r"^[\s\*]+", "", text)
        return text.strip()

    lines = raw_data.split("\n")
    header = lines[0]
    
    # Parse header để lấy BSSID và Interface
    bssid_match = re.search(r"BSS\s+([0-9a-f:]{17})", header, re.IGNORECASE)
    interface_match = re.search(r"on\s+([^\s\)]+)", header)
    
    bss = {
        "bssid": bssid_match.group(1) if bssid_match else None,
        "interface": interface_match.group(1) if interface_match else None,
        "associated": "associated" in header,
        "details": {},
    }

    # Stack dùng để quản lý cấp độ lồng nhau (indentation)
    # Cấu trúc: (mức_khoảng_trắng, dictionary_đang_ghi)
    stack = [(-1, bss["details"])]
    
    all_lines = [l for l in lines[1:] if l.strip()]

    for i, line in enumerate(all_lines):
        # Tính độ thụt lề bằng số khoảng trắng/tab ở đầu dòng
        indent = len(line) - len(line.lstrip())
        content = clean_text(line)
        
        # Kiểm tra dòng tiếp theo có thụt lề sâu hơn không để xác định đây là node cha
        has_children = False
        if i + 1 < len(all_lines):
            next_indent = len(all_lines[i + 1]) - len(all_lines[i + 1].lstrip())
            if next_indent > indent:
                has_children = True

        # Pop stack nếu dòng hiện tại thoát ra khỏi cấp độ lồng trước đó
        while len(stack) > 1 and indent <= stack[-1][0]:
            stack.pop()
        
        current_node = stack[-1][1]

        if ":" in content:
            parts = content.split(":", 1)
            key = clean_text(parts[0])
            val = clean_text(parts[1])

            # Nếu là node cha (có con hoặc thuộc danh sách đặc biệt)
            special_keys = ["WPA", "RSN", "Country", "HT capabilities", "VHT capabilities", "WMM", "Extended capabilities"]
            if has_children or key in special_keys or not val:
                new_node = {}
                if val:
                    if "Environment" in val: # Xử lý riêng cho dòng Country
                        env_parts = val.split("Environment:")
                        new_node["Code"] = clean_text(env_parts[0])
                        new_node["Environment"] = clean_text(env_parts[1])
                    else:
                        new_node["info"] = val
                
                current_node[key] = new_node
                stack.append((indent, new_node))
            else:
                # Key-Value bình thường
                current_node[key] = val
        else:
            # Nếu là dòng không có dấu ":" (thường là các flags/capabilities)
            # Nếu node hiện tại là Country thì cho vào 'constraints', ngược lại cho vào 'flags'
            target_key = "constraints" if "Code" in current_node else "flags"
            if target_key not in current_node:
                current_node[target_key] = []
            current_node[target_key].append(content)

    return bss
